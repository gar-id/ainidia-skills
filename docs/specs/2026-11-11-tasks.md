### TASK-SK-017: Full repo validation pass

**What**: Automated check that every file validates, the manifest is complete, and the repo is consistent.

**Verification** (run from the repository root; requires `ajv-cli`):
```bash
set -euo pipefail

# 1. Schemas must compile against the draft-07 meta-schema.
#    Do NOT use `ajv validate -d <schema-file>` — that validates the schema
#    document as a data instance against itself and always fails.
for s in skill persona manifest; do
  ajv compile -s "schema/$s.schema.json" >/dev/null
  echo "compile OK: $s"
done

# 2. Every content file must validate. `find` is authoritative;
#    plain `skills/**/*.json` only recurses with `shopt -s globstar`.
find skills memory -name '*.json' -print0 | while IFS= read -r -d '' f; do
  ajv validate -s schema/skill.schema.json -d "$f" >/dev/null
  echo "OK: $f"
done

for f in personas/*.json; do
  ajv validate -s schema/persona.schema.json -d "$f" >/dev/null
  echo "OK: $f"
done

ajv validate -s schema/manifest.schema.json -d manifest.json >/dev/null
echo "OK: manifest.json"

# 3. Counts: 9 skills + 1 memory = 10, plus 2 personas.
test "$(find skills memory -name '*.json' | wc -l)" -eq 10
test "$(find personas -name '*.json' | wc -l)" -eq 2
echo "counts OK: 10 skill/memory, 2 personas"

# 4. Manifest must reference exactly the files that exist (ids, paths, categories).
python3 - <<'EOF'
import json, pathlib, sys
root = pathlib.Path('.')
man = json.loads((root / 'manifest.json').read_text())
errors, seen, ids = [], set(), {}

REQUIRED_FOR = {
    'MODE_1': 'correction-immediate',
    'MODE_2': 'correction-reply-then-fix',
    'MODE_3': 'conversation-natural',
}

for section in ('personas', 'skills', 'memory'):
    for entry in man[section]:
        eid = entry['id']
        if eid in ids:
            errors.append(f"duplicate id in manifest: {eid} ({ids[eid]} and {section})")
        ids[eid] = section

        path = root / entry['path']
        if not path.exists():
            errors.append(f"{eid}: path not found -> {entry['path']}")
            continue
        doc = json.loads(path.read_text())
        if doc['id'] != eid:
            errors.append(f"{eid}: id mismatch in {entry['path']} (file says {doc['id']})")
        if doc['version'] != entry['version']:
            errors.append(f"{eid}: version mismatch ({doc['version']} vs {entry['version']})")
        if section == 'skills' and doc['category'] != entry['category']:
            errors.append(f"{eid}: category mismatch ({doc['category']} vs {entry['category']})")
        if section == 'skills' and doc['category'] == 'TOPIC' and not doc.get('starter_questions'):
            errors.append(f"{eid}: topic has no starter_questions")
        prompt_field = 'system_identity' if section == 'personas' else 'system_prompt'
        if not doc.get(prompt_field, '').strip():
            errors.append(f"{eid}: empty {prompt_field}")
        seen.add(path.resolve())

# Rule 4: each correction mode must have its mandatory skill present,
# non-deprecated, and declaring that mode in requiredFor.
# Other skills MAY also be requiredFor a mode; only the canonical one is mandatory.
present = {e['id']: e for e in man['skills'] if not e.get('deprecated')}
for mode, required_id in REQUIRED_FOR.items():
    entry = present.get(required_id)
    if entry is None:
        errors.append(f"{mode}: mandatory skill missing or deprecated -> {required_id}")
    elif mode not in entry.get('requiredFor', []):
        errors.append(f"{required_id}: must declare requiredFor {mode}")

actual = {p.resolve() for p in root.glob('**/*.json')
          if p.name != 'manifest.json' and 'schema' not in p.parts}
for extra in sorted(actual - seen):
    errors.append(f"file not listed in manifest: {extra.relative_to(root.resolve())}")

if errors:
    print('\n'.join(errors))
    sys.exit(1)
print(f"manifest OK: {len(seen)} files referenced, no orphans, no duplicate ids")
EOF
```
